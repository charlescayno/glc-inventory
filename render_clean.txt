5002: function voidSale(historyId) {
5003:     const historyIndex = historyData.findIndex(h => h.id == historyId);
5004:     if (historyIndex === -1) return;
5005: 
5006:     const rawEntry = historyData[historyIndex];
5007:     const entry = parseHistoryEntry(rawEntry);
5008:     if (entry.type !== 'sale' || entry.isVoided) return;
5009: 
5010:     if (!confirm("Are you sure you want to void this sale? This will refund the money and return the items to stock.")) {
5011:         return;
5012:     }
5013: 
5014:     // Determine location from title
5015:     let locKey = 'booth';
5016:     let titleStr = (entry.title || '').toUpperCase();
5017:     if (titleStr.includes('5TH FLOOR')) locKey = 'floor5';
5018:     else if (titleStr.includes('7TH FLOOR')) locKey = 'floor7';
5019:     
5020:     const locNames = { floor5: '5th Floor', floor7: '7th Floor', booth: 'GLC Booth' };
5021: 
5022:     // Parse items to return stock
5023:     const items = entry.items || [];
5024:     let itemsReturned = [];
5025:     
5026:     items.forEach(desc => {
5027:         const qtyMatch = typeof desc === 'string' ? desc.match(/^(\d+)x\s+(.*)/) : null;
5028:         let qty = 1;
5029:         let itemDesc = desc;
5030:         if (qtyMatch) {
5031:             qty = parseInt(qtyMatch[1]);

5033:         }
5034:         
5035:         const itemIndex = inventoryData.findIndex(i => i.desc.toLowerCase() === itemDesc.toLowerCase());
5036:         if (itemIndex !== -1) {
